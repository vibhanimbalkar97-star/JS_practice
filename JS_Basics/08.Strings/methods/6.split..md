### Split by space

```
let text = "Hello World";

console.log(text.split(" "));
// ["Hello", "World"]
```

### Split by comma

```
let text = "React,Python,JavaScript";

console.log(text.split(","));
// ["React", "Python", "JavaScript"]
```

### Split into characters

```
let text = "Hello";

console.log(text.split(""));
// ["H", "e", "l", "l", "o"]
```

### Split without a separator

```
let text = "Hello";

console.log(text.split());
// ["Hello"]
```

### Split with a limit

```
let text = "A-B-C-D";

console.log(text.split("-", 2));
// ["A", "B"]
```

### Trick

```
let text = "Hello World";

console.log(text.split(""));
// ["H", "e", "l", "l", "o", " ", "W", "o", "r", "l", "d"]
```

The space is also included as an array element.

### Important: `split()` and `join()`

```
let text = "Hello World";

let arr = text.split(" ");
console.log(arr); 
// ["Hello", "World"]

let result = arr.join("-");
console.log(result);
// Hello-World
```
